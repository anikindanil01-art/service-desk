  54  def login_required(view):
  55      @wraps(view)
  56      def wrapper(*args, **kwargs):
  57          header = request.headers.get("Authorization", "")
  58          if not header.startswith("Bearer "):
  59              raise AuthError("Требуется токен авторизации")
  60          try:
  61              payload = decode_token(header[len("Bearer "):], current_app.config["SECRET_KEY"])
  62          except jwt.PyJWTError:
  63              raise AuthError("Недействительный или просроченный токен")
  64          user = UserRepository(get_db()).get_by_id(int(payload["sub"]))
  65          if user is None:
  66              raise AuthError("Пользователь не найден")
  67          g.current_user = user
  68          return view(*args, **kwargs)
  69  
  70      return wrapper

   1  """Простой запуск SQL-миграций из каталога migrations/.
   2  
   3  Применённые миграции записываются в таблицу schema_migrations,
   4  повторный запуск применяет только новые файлы.
   5  Запуск: python -m app.migrate
   6  """
   7  from pathlib import Path
   8  
   9  from app.config import Config
  10  from app.db import connect
  11  from app.models import utcnow_iso
  12  
  13  MIGRATIONS_DIR = Path(__file__).resolve().parent.parent / "migrations"
  14  
  15  
  16  def run_migrations(conn, directory=MIGRATIONS_DIR):
  17      conn.execute(
  18          "CREATE TABLE IF NOT EXISTS schema_migrations ("
  19          "version TEXT PRIMARY KEY, applied_at TEXT NOT NULL)"
  20      )
  21      applied = {row[0] for row in conn.execute("SELECT version FROM schema_migrations")}
  22      done = []
  23      for path in sorted(Path(directory).glob("*.sql")):
  24          if path.name in applied:
  25              continue
  26          conn.executescript(path.read_text(encoding="utf-8"))
  27          conn.execute(
  28              "INSERT INTO schema_migrations (version, applied_at) VALUES (?, ?)",
  29              (path.name, utcnow_iso()),
  30          )
  31          conn.commit()
  32          done.append(path.name)
  33      return done
  34  
  35  
  36  if __name__ == "__main__":
  37      connection = connect(Config.DATABASE)
  38      result = run_migrations(connection)
  39      connection.close()
  40      if result:
  41          print("Применены миграции:", ", ".join(result))
  42      else:
  43          print("Новых миграций нет")

   9  def hash_password(password, iterations):
  10      salt = os.urandom(16)
  11      digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, iterations)
  12      return f"{iterations}${salt.hex()}${digest.hex()}"
  13  
  14  
  15  def verify_password(password, stored):
  16      try:
  17          iterations, salt_hex, digest_hex = stored.split("$")
  18          digest = hashlib.pbkdf2_hmac(
  19              "sha256", password.encode("utf-8"), bytes.fromhex(salt_hex), int(iterations)
  20          )
  21      except ValueError:
  22          return False
  23      return hmac.compare_digest(digest.hex(), digest_hex)

  30          "exp": datetime.now() + timedelta(minutes=expires_minutes),
  31      }
  32      return jwt.encode(payload, secret, algorithm="HS256")

  15  class AuthService:
  16      def __init__(self, users, secret, expires_minutes, iterations):
  17          self.users = users
  18          self.secret = secret
  19          self.expires_minutes = expires_minutes
  20          self.iterations = iterations
  21  
  22      def register(self, data):
  23          username = data.get("username")
  24          password = data.get("password")
  25          role = data.get("role", ROLE_USER)
  26          if not isinstance(username, str) or not 3 <= len(username.strip()) <= 50:
  27              raise ValidationError("username должен содержать от 3 до 50 символов")
  28          if not isinstance(password, str) or len(password) < 6:
  29              raise ValidationError("password должен содержать не менее 6 символов")
  30          if role not in ROLES:
  31              raise ValidationError("Недопустимая роль, допустимы: " + ", ".join(ROLES))
  32          username = username.strip()
  33          if self.users.get_by_username(username) is not None:
  34              raise ConflictError("Пользователь с таким именем уже существует")
  35          password_hash = hash_password(password, self.iterations)
  36          user = self.users.create(username, password_hash, role, utcnow_iso())
  37          return public_user(user)
  38  
  39      def login(self, data):
  40          username = data.get("username")
  41          password = data.get("password")
  42          if not isinstance(username, str) or not isinstance(password, str):
  43              raise ValidationError("Нужно передать username и password")
  44          user = self.users.get_by_username(username.strip())
  45          if user is None or not verify_password(password, user["password_hash"]):
  46              raise AuthError("Неверное имя пользователя или пароль")
  47          token = create_token(user["id"], user["role"], self.secret, self.expires_minutes)
  48          return {"access_token": token, "token_type": "Bearer"}
